cmake_minimum_required(VERSION 3.24)
project(ChengVault VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
option(CHENGVAULT_COVERAGE "Instrument project code for gcov" OFF)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Widgets Network Concurrent Test)
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS src/core/*.cpp src/platform/*.cpp)
add_library(chengvault_core STATIC ${CORE_SOURCES})
target_include_directories(chengvault_core PUBLIC include)
target_link_libraries(chengvault_core PUBLIC Qt6::Core Qt6::Network Qt6::Concurrent)
target_compile_definitions(chengvault_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
if(WIN32)
  target_link_libraries(chengvault_core PUBLIC advapi32 bcrypt crypt32 ws2_32)
endif()
if(MINGW)
  target_compile_options(chengvault_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wno-sign-conversion)
endif()
if(CHENGVAULT_COVERAGE AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  add_compile_options(--coverage -O0 -g)
  add_link_options(--coverage)
  target_compile_options(chengvault_core PRIVATE --coverage -O0 -g)
  target_link_options(chengvault_core PUBLIC --coverage)
endif()
foreach(pair "ChengVaultCli;cli_main.cpp" "ChengVaultAgent;agent_main.cpp" "ChengVaultServer;server_main.cpp")
  list(GET pair 0 target)
  list(GET pair 1 source)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/apps/${source}")
    add_executable(${target} src/apps/${source})
    target_link_libraries(${target} PRIVATE chengvault_core)
  endif()
endforeach()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/apps/gui_main.cpp")
  file(GLOB GUI_SOURCES CONFIGURE_DEPENDS src/gui/*.cpp src/gui/*.hpp)
  add_executable(ChengVault WIN32 src/apps/gui_main.cpp ${GUI_SOURCES})
  target_link_libraries(ChengVault PRIVATE chengvault_core Qt6::Widgets Qt6::Concurrent)
endif()
include(CTest)
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_main.cpp")
  file(GLOB TEST_SOURCES CONFIGURE_DEPENDS tests/*_tests.cpp)
  add_executable(ChengVaultTests tests/test_main.cpp ${TEST_SOURCES})
  target_include_directories(ChengVaultTests PRIVATE tests)
  target_link_libraries(ChengVaultTests PRIVATE chengvault_core Qt6::Test)
  add_test(NAME ChengVaultFunctional COMMAND ChengVaultTests)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/background_process.cpp")
    add_executable(ChengVaultBackgroundTests tests/background_process.cpp)
    target_include_directories(ChengVaultBackgroundTests PRIVATE tests)
    target_link_libraries(ChengVaultBackgroundTests PRIVATE chengvault_core Qt6::Test)
    add_test(NAME ChengVaultBackgroundProcess COMMAND ChengVaultBackgroundTests)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/gui_smoke.cpp")
    add_executable(ChengVaultGuiTests tests/gui_smoke.cpp ${GUI_SOURCES})
    target_include_directories(ChengVaultGuiTests PRIVATE tests src/gui)
    target_link_libraries(ChengVaultGuiTests PRIVATE chengvault_core Qt6::Widgets Qt6::Test)
    add_test(NAME ChengVaultGuiSmoke COMMAND ChengVaultGuiTests)
  endif()
endif()
